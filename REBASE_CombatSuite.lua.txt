local function IsFiniteNumber(n: any): boolean
    return typeof(n) == "number" and n == n and math.abs(n) < math.huge
end

local function IsFiniteVector3(v: any): boolean
    return typeof(v) == "Vector3"
        and IsFiniteNumber(v.X)
        and IsFiniteNumber(v.Y)
        and IsFiniteNumber(v.Z)
end

local function GetSafeGuiParent(): Instance

end

-- Character & entity validation helpers
local function GetCharacterParts(character: Model?)
    if not character then
        return nil, nil
    end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart") :: BasePart?
    if not humanoid or not rootPart or humanoid.Health <= 0 then
        return nil, nil
    end
    return humanoid, rootPart
end

local function ResolveTargetPart(character: Model, preferredName: string): BasePart?
    local part = character:FindFirstChild(preferredName)
    if part and part:IsA("BasePart") then
        return part
    end
    local fallback = character:FindFirstChild("HumanoidRootPart")
    if fallback and fallback:IsA("BasePart") then
        return fallback
    end
    return nil
end

local function IsSameTeam(player: Player): boolean
    if not LocalPlayer then
        return false
    end
    if LocalPlayer.Team ~= nil and player.Team ~= nil then
        return LocalPlayer.Team == player.Team
    end
    return false
end

local function IsPositionVisible(origin: Vector3, targetPos: Vector3, targetCharacter: Model): boolean
    local direction = targetPos - origin
    if direction.Magnitude <= 0.001 then
        return true
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local ignore = {}
    if LocalPlayer and LocalPlayer.Character then
        table.insert(ignore, LocalPlayer.Character)
    end
    if Camera then
        table.insert(ignore, Camera)
    end
    params.FilterDescendantsInstances = ignore
    params.IgnoreWater = true

    local result = Workspace:Raycast(origin, direction, params)
    if not result or not result.Instance then
        return true
    end
    return result.Instance:IsDescendantOf(targetCharacter)
end

-- Network latency & velocity prediction
local function GetPingSeconds(): number
    local ok, pingMs = pcall(function()
        local network = Stats:FindFirstChild("Network")
        local serverStats = network and network:FindFirstChild("ServerStatsItem")
        local dataPing = serverStats and serverStats:FindFirstChild("Data Ping")
        return dataPing and dataPing:GetValue() or 0
    end)
    if not ok or not IsFiniteNumber(pingMs) or pingMs <= 0 then
        return 0.05
    end
    local clamped = math.clamp(pingMs, Config.Prediction.PingFloorMs, Config.Prediction.PingCeilMs)
    return clamped / 1000
end

local function GetCurrentLeadFactors(): (number, number)
    if not Config.Prediction.Enabled then
        return 0, 0
    end
    if Config.Prediction.AutoPrediction then
        local pingSec = GetPingSeconds()
        local baseLead = (pingSec + (pingSec * pingSec * 0.35)) * Config.Prediction.AutoScale
        return baseLead, baseLead * 0.75
    end
    return Config.Prediction.ManualHorizontal, Config.Prediction.ManualVertical
end

local function GetPredictedPosition(targetPart: BasePart, rootPart: BasePart): Vector3
    local basePos = targetPart.Position
    if not Config.Prediction.Enabled then
        return basePos
    end

    local velocity = rootPart.AssemblyLinearVelocity
    if not IsFiniteVector3(velocity) or velocity.Magnitude < 0.05 then
        return basePos
    end

    local leadH, leadV = GetCurrentLeadFactors()
    local predicted = basePos + Vector3.new(
        velocity.X * leadH,
        velocity.Y * leadV,
        velocity.Z * leadH
    )
    if IsFiniteVector3(predicted) then
        return predicted
    end
    return basePos
end

-- FOV origin & target selection
local function GetFOVOrigin(): Vector2
    if Config.FOV.FollowMouse and not IsTouchDevice then
        return UserInputService:GetMouseLocation()
    end
    Camera = Workspace.CurrentCamera
    if Camera then
        return Camera.ViewportSize * 0.5
    end
    return Vector2.zero
end

local function ValidateSpecificTarget(
    player: Player?,
    partName: string,
    useFOV: boolean,
    teamCheck: boolean,
    wallCheck: boolean
): (BasePart?, BasePart?)
    if not player or player == LocalPlayer or not player.Parent then
        return nil, nil
    end
    if teamCheck and IsSameTeam(player) then
        return nil, nil
    end

    local character = player.Character
    if not character then
        return nil, nil
    end

    local _, rootPart = GetCharacterParts(character)
    if not rootPart then
        return nil, nil
    end

    local targetPart = ResolveTargetPart(character, partName)
    if not targetPart then
        return nil, nil
    end

    Camera = Workspace.CurrentCamera
    if not Camera then
        return nil, nil
    end

    local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
    if not onScreen or screenPos.Z <= 0 then
        return nil, nil
    end

    if useFOV then
        local fovOrigin = GetFOVOrigin()
        local dist = (Vector2.new(screenPos.X, screenPos.Y) - fovOrigin).Magnitude
        if dist > Config.FOV.Radius then
            return nil, nil
        end
    end

    if wallCheck and not IsPositionVisible(Camera.CFrame.Position, targetPart.Position, character) then
        return nil, nil
    end

    return targetPart, rootPart
end

local function GetBestTarget(
    partName: string,
    useFOV: boolean,
    teamCheck: boolean,
    wallCheck: boolean
): (Player?, BasePart?, BasePart?)
    Camera = Workspace.CurrentCamera
    if not Camera then
        return nil, nil, nil
    end

    local fovOrigin = GetFOVOrigin()
    local bestPlayer: Player? = nil
    local bestPart: BasePart? = nil
    local bestRoot: BasePart? = nil
    local bestDistance = useFOV and Config.FOV.Radius or math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            if not (teamCheck and IsSameTeam(player)) then
                local character = player.Character
                local _, rootPart = GetCharacterParts(character)
                if character and rootPart then
                    local targetPart = ResolveTargetPart(character, partName)
                    if targetPart then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                        if onScreen and screenPos.Z > 0 then
                            local screenDist = (Vector2.new(screenPos.X, screenPos.Y) - fovOrigin).Magnitude
                            if screenDist <= bestDistance then
                                if not wallCheck or IsPositionVisible(Camera.CFrame.Position, targetPart.Position, character) then
                                    bestDistance = screenDist
                                    bestPlayer = player
                                    bestPart = targetPart
                                    bestRoot = rootPart
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    return bestPlayer, bestPart, bestRoot
end

-- Mobile Touch HUD & GUI FOV Fallback
local function UpdateTouchHUDVisuals()
    local activeColor = Color3.fromRGB(0, 175, 225)
    local inactiveColor = Color3.fromRGB(28, 32, 42)
    local alpha = math.clamp(Config.Mobile.ButtonTransparency, 0, 0.85)

    local states = {
        Aim = Config.Aimbot.Enabled and (State.AimbotToggled or State.TouchAimHeld),
        Silent = Config.SilentAim.Enabled,
        Speed = Config.SpeedWalk.Enabled,
        Desync = Config.Desync.Enabled,
    }

    for key, btn in pairs(State.TouchButtons) do
        if btn and btn:IsA("TextButton") then
            local isOn = states[key] == true
            btn.BackgroundColor3 = isOn and activeColor or inactiveColor
            btn.BackgroundTransparency = isOn and math.max(0.05, alpha - 0.1) or alpha
            btn.TextColor3 = isOn and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(210, 218, 230)
        end
    end
end

-- Silent Aim Raycast Metamethod Hook (__namecall)
local function InstallSilentAimHook()
    if not CanHookNamecall or State.NamecallHooked then
        return
    end

    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        if State.Unloaded or checkcaller() or not Config.SilentAim.Enabled then
            return oldNamecall(self, ...)
        end

        if self == Workspace then
            local method = getnamecallmethod()
            local selectedMode = Config.SilentAim.Method

            local handleModern = (method == "Raycast")
                and (selectedMode == "Workspace:Raycast" or selectedMode == "All Raycast")
            local handleLegacy = (
                method == "FindPartOnRay"
                or method == "FindPartOnRayWithIgnoreList"
                or method == "FindPartOnRayWithWhitelist"
            ) and (selectedMode == "FindPartOnRay" or selectedMode == "All Raycast")

            if (handleModern or handleLegacy) and math.random(1, 100) <= Config.SilentAim.HitChance then
                local args = { ... }

                if handleModern then
                    local origin = args[1]
                    local direction = args[2]
                    if IsFiniteVector3(origin) and IsFiniteVector3(direction) and direction.Magnitude > 0.001 then
                        local ok, _, targetPart, targetRoot = pcall(
                            GetBestTarget,
                            Config.SilentAim.TargetPart,
                            Config.SilentAim.UseFOV,
                            Config.SilentAim.TeamCheck,
                            Config.SilentAim.WallCheck
                        )
                        if ok and targetPart and targetRoot then
                            local predictedPos = GetPredictedPosition(targetPart, targetRoot)
                            local delta = predictedPos - origin
                            if delta.Magnitude > 0.001 then
                                args[2] = delta.Unit * direction.Magnitude
                                return oldNamecall(self, table.unpack(args))
                            end
                        end
                    end
                elseif handleLegacy then
                    local ray = args[1]
                    if typeof(ray) == "Ray" and ray.Direction.Magnitude > 0.001 then
                        local ok, _, targetPart, targetRoot = pcall(
                            GetBestTarget,
                            Config.SilentAim.TargetPart,
                            Config.SilentAim.UseFOV,
                            Config.SilentAim.TeamCheck,
                            Config.SilentAim.WallCheck
                        )
                        if ok and targetPart and targetRoot then
                            local predictedPos = GetPredictedPosition(targetPart, targetRoot)
                            local delta = predictedPos - ray.Origin
                            if delta.Magnitude > 0.001 then
                                args[1] = Ray.new(ray.Origin, delta.Unit * ray.Direction.Magnitude)
                                return oldNamecall(self, table.unpack(args))
                            end
                        end
                    end
                end
            end
        end

        return oldNamecall(self, ...)
    end))

    State.NamecallHooked = true
end

-- Aimbot activation check (supports desktop hold/toggle AND mobile touch button)
local function IsAimbotActive(): boolean
    if not Config.Aimbot.Enabled then
        return false
    end
    if Config.Mobile.Enabled then
        if Config.Mobile.TouchAimHoldMode and State.TouchAimHeld then
            return true
        end
        if not Config.Mobile.TouchAimHoldMode and State.AimbotToggled then
            return true
        end
    end
    if Config.Aimbot.ToggleMode then
        return State.AimbotToggled
    end
    local key = Config.Aimbot.Key
    if typeof(key) == "EnumItem" then
        if key.EnumType == Enum.UserInputType then
            return UserInputService:IsMouseButtonPressed(key :: Enum.UserInputType)
        elseif key.EnumType == Enum.KeyCode then
            return UserInputService:IsKeyDown(key :: Enum.KeyCode)
        end
    end
    return false
end

TrackConnection(UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed or State.Unloaded then
        return
    end
    if Config.Aimbot.Enabled and Config.Aimbot.ToggleMode then
        if input.UserInputType == Config.Aimbot.Key or input.KeyCode == Config.Aimbot.Key then
            State.AimbotToggled = not State.AimbotToggled
            if not State.AimbotToggled then
                State.LockedTarget = nil
            end
            UpdateTouchHUDVisuals()
        end
    end
end))

-- Restore helpers for clean disable/unload
local function RestoreCharacterState()
    local character = LocalPlayer and LocalPlayer.Character
    local humanoid, rootPart = GetCharacterParts(character)

    if humanoid then
        if State.SavedWalkSpeed ~= nil then
            humanoid.WalkSpeed = State.SavedWalkSpeed
            State.SavedWalkSpeed = nil
        end
        if State.SavedAutoRotate ~= nil then
            humanoid.AutoRotate = State.SavedAutoRotate
            State.SavedAutoRotate = nil
        end
    end

    if rootPart then
        if State.DesyncActiveOffset.Magnitude > 0 then
            rootPart.CFrame = rootPart.CFrame - State.DesyncActiveOffset
            State.DesyncActiveOffset = Vector3.zero
        end
        if State.SavedVelocity ~= nil then
            rootPart.AssemblyLinearVelocity = State.SavedVelocity
            State.SavedVelocity = nil
        end
        local rootJoint = rootPart:FindFirstChild("RootJoint")
        if rootJoint and rootJoint:IsA("Motor6D") then
            rootJoint.Transform = CFrame.new()
        end
    end

    if State.FFlagModified and Capabilities.SetFFlag then
        pcall(setfflag, "NextGenReplicatorEnabledWrite4", "false")
        State.FFlagModified = false
    end
end

-- Pre-render step: restore desynced local visual transform, update FOV circle & Aimbot
TrackConnection(RunService.RenderStepped:Connect(function(dt: number)
    if State.Unloaded then
        return
    end

    Camera = Workspace.CurrentCamera
    local character = LocalPlayer and LocalPlayer.Character
    local _, rootPart = GetCharacterParts(character)

    if rootPart then
        if State.DesyncActiveOffset.Magnitude > 0 then
            rootPart.CFrame = rootPart.CFrame - State.DesyncActiveOffset
            State.DesyncActiveOffset = Vector3.zero
        end
        if State.SavedVelocity ~= nil then
            rootPart.AssemblyLinearVelocity = State.SavedVelocity
            State.SavedVelocity = nil
        end
    end

    -- Update FOV Circle (Drawing API or Mobile GUI Fallback)
    local fovOrigin = GetFOVOrigin()
    local useGuiFOV = Config.Mobile.UseGuiFOVFallback or not State.FOVCircle

    if State.FOVCircle then
        local showDrawing = Config.FOV.Enabled and not useGuiFOV
        State.FOVCircle.Visible = showDrawing
        if showDrawing then
            State.FOVCircle.Position = fovOrigin
            State.FOVCircle.Radius = Config.FOV.Radius
            State.FOVCircle.Thickness = Config.FOV.Thickness
            State.FOVCircle.Filled = Config.FOV.Filled
            State.FOVCircle.Transparency = Config.FOV.Transparency
            State.FOVCircle.Color = Config.FOV.Color
        end
    end

    if State.GuiFOVFrame and State.GuiFOVStroke then
        local showGui = Config.FOV.Enabled and useGuiFOV
        State.GuiFOVFrame.Visible = showGui
        if showGui then
            local diameter = Config.FOV.Radius * 2
            State.GuiFOVFrame.Position = UDim2.fromOffset(fovOrigin.X, fovOrigin.Y)
            State.GuiFOVFrame.Size = UDim2.fromOffset(diameter, diameter)
            State.GuiFOVFrame.BackgroundColor3 = Config.FOV.Color
            State.GuiFOVFrame.BackgroundTransparency = Config.FOV.Filled and (1 - Config.FOV.Transparency) or 1
            State.GuiFOVStroke.Thickness = Config.FOV.Thickness
            State.GuiFOVStroke.Color = Config.FOV.Color
            State.GuiFOVStroke.Transparency = 1 - Config.FOV.Transparency
        end
    end

    -- Execute Camera Aimbot
    if IsAimbotActive() and Camera then
        local targetPart: BasePart? = nil
        local targetRoot: BasePart? = nil

        if Config.Aimbot.StickyTarget and State.LockedTarget then
            targetPart, targetRoot = ValidateSpecificTarget(
                State.LockedTarget,
                Config.Aimbot.TargetPart,
                Config.FOV.Enabled,
                Config.Aimbot.TeamCheck,
                Config.Aimbot.WallCheck
            )
            if not targetPart then
                State.LockedTarget = nil
            end
        end

        if not targetPart then
            local foundPlayer, foundPart, foundRoot = GetBestTarget(
                Config.Aimbot.TargetPart,
                Config.FOV.Enabled,
                Config.Aimbot.TeamCheck,
                Config.Aimbot.WallCheck
            )
            State.LockedTarget = foundPlayer
            targetPart = foundPart
            targetRoot = foundRoot
        end

        if targetPart and targetRoot then
            local predictedPos = GetPredictedPosition(targetPart, targetRoot)
            local camPos = Camera.CFrame.Position
            if (predictedPos - camPos).Magnitude > 0.01 then
                local goalCFrame = CFrame.lookAt(camPos, predictedPos)
                local alpha = 1 - math.pow(1 - math.clamp(Config.Aimbot.Smoothness, 0.02, 1), math.clamp(dt * 60, 0.01, 4))
                Camera.CFrame = Camera.CFrame:Lerp(goalCFrame, alpha)
            end
        end
    else
        State.LockedTarget = nil
    end
end))

-- Post-physics Heartbeat step: Speed Walk, Anti-Aim, and Replication Desync
TrackConnection(RunService.Heartbeat:Connect(function(dt: number)
    if State.Unloaded then
        return
    end

    local character = LocalPlayer and LocalPlayer.Character
    local humanoid, rootPart = GetCharacterParts(character)
    if not humanoid or not rootPart then
        State.DesyncActiveOffset = Vector3.zero
        State.SavedVelocity = nil
        return
    end

    -- 1. Speed Walk
    if Config.SpeedWalk.Enabled then
        if Config.SpeedWalk.Method == "WalkSpeed" then
            if State.SavedWalkSpeed == nil then
                State.SavedWalkSpeed = humanoid.WalkSpeed
            end
            humanoid.WalkSpeed = Config.SpeedWalk.Speed
        else
            if State.SavedWalkSpeed ~= nil then
                humanoid.WalkSpeed = State.SavedWalkSpeed
                State.SavedWalkSpeed = nil
            end
            local moveDir = humanoid.MoveDirection
            local extraSpeed = Config.SpeedWalk.Speed - Config.SpeedWalk.DefaultWalkSpeed
            if moveDir.Magnitude > 0.01 and extraSpeed ~= 0 then
                rootPart.CFrame = rootPart.CFrame + (moveDir.Unit * extraSpeed * dt)
            end
        end
    elseif State.SavedWalkSpeed ~= nil then
        humanoid.WalkSpeed = State.SavedWalkSpeed
        State.SavedWalkSpeed = nil
    end

    -- 2. Anti-Aim
    if Config.AntiAim.Enabled then
        if State.SavedAutoRotate == nil then
            State.SavedAutoRotate = humanoid.AutoRotate
        end
        humanoid.AutoRotate = false

        Camera = Workspace.CurrentCamera
        local baseYaw = 0
        if Camera then
            local look = Camera.CFrame.LookVector
            baseYaw = math.atan2(-look.X, -look.Z)
        end

        local offsetYaw = 0
        local mode = Config.AntiAim.YawMode
        if mode == "Spin" then
            State.SpinAngle = (State.SpinAngle + math.rad(Config.AntiAim.SpinSpeed * 30) * dt) % (math.pi * 2)
            offsetYaw = State.SpinAngle
        elseif mode == "Jitter" then
            State.JitterSign = -State.JitterSign
            offsetYaw = math.rad(Config.AntiAim.JitterAngle) * State.JitterSign
        elseif mode == "Backward" then
            offsetYaw = math.pi
        elseif mode == "Random" then
            offsetYaw = math.rad(math.random(-180, 180))
        end

        local pos = rootPart.Position
        rootPart.CFrame = CFrame.new(pos) * CFrame.Angles(0, baseYaw + offsetYaw, 0)

        local rootJoint = rootPart:FindFirstChild("RootJoint")
        if rootJoint and rootJoint:IsA("Motor6D") then
            if Config.AntiAim.PitchTilt ~= 0 then
                rootJoint.Transform = CFrame.Angles(math.rad(Config.AntiAim.PitchTilt), 0, 0)
            else
                rootJoint.Transform = CFrame.new()
            end
        end
    elseif State.SavedAutoRotate ~= nil then
        humanoid.AutoRotate = State.SavedAutoRotate
        State.SavedAutoRotate = nil
        local rootJoint = rootPart:FindFirstChild("RootJoint")
        if rootJoint and rootJoint:IsA("Motor6D") then
            rootJoint.Transform = CFrame.new()
        end
    end

    -- 3. Desync
    if Config.Desync.Enabled then
        local desyncMode = Config.Desync.Mode
        if desyncMode == "CFrame Blink" then
            if State.FFlagModified and Capabilities.SetFFlag then
                pcall(setfflag, "NextGenReplicatorEnabledWrite4", "false")
                State.FFlagModified = false
            end
            local range = Config.Desync.OffsetRange
            local ox = (math.random() * 2 - 1) * range
            local oz = (math.random() * 2 - 1) * range
            local offset = Vector3.new(ox, Config.Desync.VerticalOffset, oz)
            rootPart.CFrame = rootPart.CFrame + offset
            State.DesyncActiveOffset = offset
        elseif desyncMode == "Velocity Spoof" then
            if State.FFlagModified and Capabilities.SetFFlag then
                pcall(setfflag, "NextGenReplicatorEnabledWrite4", "false")
                State.FFlagModified = false
            end
            State.SavedVelocity = rootPart.AssemblyLinearVelocity
            local mag = Config.Desync.VelocityMagnitude
            rootPart.AssemblyLinearVelocity = Vector3.new(
                (math.random() * 2 - 1) * mag,
                State.SavedVelocity.Y,
                (math.random() * 2 - 1) * mag
            )
        elseif desyncMode == "FFlag Replication" and Capabilities.SetFFlag then
            local flagVal = (math.floor(os.clock() * 20) % 2 == 0) and "true" or "false"
            pcall(setfflag, "NextGenReplicatorEnabledWrite4", flagVal)
            State.FFlagModified = true
        end
    else
        if State.FFlagModified and Capabilities.SetFFlag then
            pcall(setfflag, "NextGenReplicatorEnabledWrite4", "false")
            State.FFlagModified = false
        end
    end
end))

-- Unified unload / teardown routine
local function UnloadSuite()
    if State.Unloaded then
        return
    end
    State.Unloaded = true
    Config.Aimbot.Enabled = false
    Config.SilentAim.Enabled = false
    Config.SpeedWalk.Enabled = false
    Config.Desync.Enabled = false
    Config.AntiAim.Enabled = false
    State.TouchAimHeld = false

    for _, conn in ipairs(State.Connections) do
        if conn.Connected then
            conn:Disconnect()
        end
    end
    table.clear(State.Connections)

    if State.FOVCircle then
        pcall(function()
            State.FOVCircle:Remove()
        end)
        State.FOVCircle = nil
    end

    if State.TouchGui then
        pcall(function()
            State.TouchGui:Destroy()
        end)
        State.TouchGui = nil
    end

    RestoreCharacterState()
end
